\section{Planteamiento del problema}
% ============================================================

\begin{frame}{El problema: crecimiento disperso que rebasa a la planeación}
  \begin{columns}[T]
    \begin{column}{0.56\textwidth}
      \small
      \begin{itemize}
        \item México tiene instrumentos de ordenamiento territorial (LGAHOTDU, SEDATU, ONU-Hábitat), pero su cobertura efectiva es \highlight{limitada}.
        \item Solo el \keynum{20.5\%} de los municipios tiene un instrumento vigente; dos tercios de los planes no se han actualizado.
        \item Según INEGI, entre el \keynum{43\%} y el \keynum{80\%} de la población reside en zonas urbanas.
        \item Resultado: un patrón de \highlight{crecimiento disperso} que supera la capacidad de planeación.
      \end{itemize}
      \begin{alertblock}{Caso de estudio: Querétaro}
        \footnotesize Ciudad intermedia dinámica, con serie histórica larga (\textbf{37 años}) y datos de acceso libre.
      \end{alertblock}
    \end{column}
    \begin{column}{0.44\textwidth}
      \begin{center}
        \safefig[width=\linewidth]{crecimiento_disperso_zmq_1984_2020.png}\\
        \scriptsize\textcolor{Slate}{Capturas RGB de la ZMQ (1984 · 2010 · 2020): expansión periférica dispersa.}
      \end{center}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{¿Qué debe cumplir un modelo útil para la planeación?}
  \small
  Para servir como herramienta de apoyo a la planeación, el modelo debe satisfacer \highlight{cuatro requerimientos} simultáneos:
  \vspace{0.2cm}
  \begin{block}{Requerimientos R1--R4}
  \begin{tabularx}{\linewidth}{@{}p{0.7cm}>{\raggedright\arraybackslash}p{3.3cm}X@{}}
    \textbf{R1} & Simulación histórica precisa & Reproducir el cambio urbano/no-urbano; evaluado con \textbf{FoM, Kappa, IoU} en ventanas independientes. \\[2pt]
    \textbf{R2} & Soporte a planeación & Mapas y escenarios espacialmente explícitos. \\[2pt]
    \textbf{R3} & Replicabilidad y apertura & Flujo reproducible \textbf{sin software propietario} ni hardware especializado. \\[2pt]
    \textbf{R4} & Interpretabilidad & La función de transición debe ser \textbf{auditable}: pesos legibles como evidencia espacial, no parámetros opacos. \\
  \end{tabularx}
  \end{block}
  \centering\footnotesize R4 resultó ser la condición \highlight{más restrictiva} del campo.
\end{frame}

\begin{frame}{Treinta años de modelado urbano computacional}
  \small
  \begin{itemize}
    \item \highlight{1997--} Reglas locales de distancia/vecindad reproducen patrones realistas (\textit{White \& Engelen}); \textbf{SLEUTH} (\textit{Clarke}) se vuelve referencia por dos décadas.
    \item \highlight{LUCC} Mapas de idoneidad estadística: CLUE-S, Dinamica EGO, CA-Markov, FLUS, MOLUSCE (regresión logística o pesos bayesianos).
    \item \highlight{ML/DL e híbridos} Redes neuronales y metaheurísticas (\textit{Almeida, Wang, Hagenauer, Ma}) mejoran la precisión\ldots

  \end{itemize}
  \vspace{0.2cm}
  \begin{block}{La tensión real del campo}
    \centering No es ``AC vs.\ aprendizaje automático'', sino \highlight{precisión predictiva} frente a \highlight{trazabilidad de cada decisión espacial}.
  \end{block}
\end{frame}

\section{Datos y modelo WoE-AC}
% ============================================================

\begin{frame}{Preprocesamiento: de RGB a mapa binario}
  \begin{columns}[c]
    \begin{column}{0.44\textwidth}
      \begin{center}
        \safefig[width=\linewidth]{pipeline_rgb_features_binary_cap02.png}
      \end{center}
    \end{column}
    \begin{column}{0.56\textwidth}
      \footnotesize
      \begin{itemize}
        \item RGB $\rightarrow$ \highlight{23 features} por celda (textura LBP/Sobel/entropía + color LAB/HSV): NDVI/NDWI/NDBI \textbf{no} son computables sin NIR/SWIR.
        \item \highlight{StandardScaler} $\rightarrow$ \highlight{PCA} a 8 componentes ($\sim$96\% de varianza).
        \item \highlight{K-Means} ($k{=}2$): no supervisado $\Rightarrow$ \textbf{consistencia} del criterio en los 37 años.
        \item \highlight{SVM lineal} sobre pseudo-etiquetas regulariza la frontera: exactitud \keynum{88--94\%} en \emph{hold-out}.
        \item Convención urbano${=}1$ / no-urbano${=}0$.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{WoE como función de transición}
  \small
  \begin{columns}[T]
    \begin{column}{0.58\textwidth}
      \begin{itemize}
        \item \highlight{Transición}: celda no-urbana en $t$ que pasa a urbana en $t{+}1$.
        \item \highlight{WoE} en vez de regresión logística: no impone forma funcional lineal; discretiza y estima un peso bayesiano por intervalo $\Rightarrow$ captura la \textbf{no linealidad} real (caída abrupta cerca del frente urbano).
        \item \highlight{7 variables} derivadas \emph{solo} del mapa binario (sin capas externas): distancia al frente, densidades $3{\times}3/5{\times}5/7{\times}7$, fragmentación, tamaño de cluster, gradiente.
        \item \highlight{Pooling} de 26 pares (1984--2010): \keynum{$8{,}196{,}264$} transiciones $\Rightarrow$ pesos estables.
        \item Combinación ponderada por \textbf{IV} y sigmoide $\rightarrow$ probabilidad en $[0,1]$.
      \end{itemize}
    \end{column}
    \begin{column}{0.42\textwidth}
      \begin{center}
        \safefig[width=\linewidth]{woe_weights.png}\\
        \scriptsize\textcolor{Slate}{Relación monótona WoE vs.\ densidad de vecindad: valida el mecanismo de difusión.}
      \end{center}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{¿Por qué un autómata celular y no el mapa WoE estático?}
  \small
  \begin{block}{La razón}
    El mapa WoE describe la \highlight{idoneidad} de cada celda dada la configuración inicial, pero \highlight{no simula cómo se propaga} el crecimiento en el tiempo.
  \end{block}
  \begin{itemize}
    \item El AC aplica la regla de transición \highlight{iterativamente}: en cada paso anual se \textbf{recalculan} las 7 variables sobre el estado actualizado.
    \item Cuando una celda se urbaniza, eleva la densidad de sus vecinas $\Rightarrow$ \highlight{contagio borde a borde} (mancha compacta, relleno de huecos, corredores).
    \item Regla de transición = \keynum{WoE} $+$ \keynum{vecindad de Moore} (8 vecinos, peso $\alpha$) $>$ \keynum{umbral} $\theta$, con componente \highlight{estocástica}.
  \end{itemize}
  \vspace{0.1cm}
  \centering\footnotesize Que las dos variables más predictivas sean de proximidad local confirma que el crecimiento es por \highlight{contagio}, no difusión aleatoria.
\end{frame}

\begin{frame}{Definición matemática de la función de transición}
  \begin{block}{Probabilidad de transición de la celda $(i,j)$ en el paso $t$}
  \[
    P_{\text{trans}}(i,j,t)=\sigma\!\Bigg(
    \underbrace{\sum_{k=1}^{7}\frac{\textcolor{Steel}{IV_k}}{\sum_{l} IV_l}\,\mathrm{WoE}_k(i,j,t)}_{\text{evidencia espacial ponderada por IV}}
    \;+\;
    \underbrace{\textcolor{Teal}{\alpha}\,I_{\text{neigh}}(i,j,t)}_{\text{vecindad de Moore}}
    \Bigg)
  \]
  \end{block}
  \[
    \text{La celda transita a urbana si}\quad P_{\text{trans}}(i,j,t)\;>\;\textcolor{Amber}{\theta}
    \quad\text{(con confirmación estocástica).}
  \]
  \vspace{-0.15cm}
  \footnotesize
  \begin{itemize}
    \item $\sigma(\cdot)$: función sigmoide $\Rightarrow$ probabilidad acotada en $[0,1]$.
    \item $\dfrac{\textcolor{Steel}{IV_k}}{\sum_l IV_l}$: \highlight{ponderación por Information Value} normalizado; las variables más discriminativas pesan más (la distancia al frente concentra $\approx 28\%$).
    \item $\textcolor{Teal}{\alpha}=0{,}50$: \highlight{peso de la vecindad} de Moore $I_{\text{neigh}}$ (8 vecinos).
    \item $\textcolor{Amber}{\theta}=0{,}75$: \highlight{umbral} de urbanización (calibrado; valores menores sobre-predicen).
  \end{itemize}
\end{frame}

\begin{frame}{Posicionamiento frente a la literatura}
\begin{columns}[T]
  \begin{column}{0.62\textwidth}
    \small
    \begin{itemize}
      \item \textcolor{Slate}{Pontius et al.\ (2008)}: de 13 aplicaciones, \keynum{6} con FoM $<15\%$; solo \keynum{1} supera $50\%$.
      \item \textcolor{Slate}{GSA-CA Urumqi (Tang 2024)}: FoM $\approx 37$--$43\%$ con \emph{drivers} físicos explícitos (pendiente, vialidades, empleo).
      \item \highlight{Este trabajo}: FoM promedio \keynum{31{,}7\%} usando \emph{solo} 7 variables derivadas del mapa binario — sin datos externos.
    \end{itemize}
  \end{column}
  \begin{column}{0.36\textwidth}
    \begin{block}{}
      \centering\footnotesize Desempeño en la \highlight{banda intermedia} del espectro empírico global, \textbf{sin sacrificar interpretabilidad} ni requerir \emph{drivers} externos.
    \end{block}
  \end{column}
\end{columns}
\end{frame}

\begin{frame}{Análisis del error: fortalezas y debilidades}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \textbf{\textcolor{GreenOK}{Fortalezas}}
      \small
      \begin{itemize}
        \item Recall $>\keynum{0{,}91}$ en todas las ventanas: captura casi todo el crecimiento real.
        \item Fidelidad morfológica: el crecimiento se concentra en bordes (difusión contigua).
        \item Parsimonia: 7 variables, sin datos socioeconómicos externos.
        \item \highlight{$\sim$3 h en CPU}, sin GPU.
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \textbf{\textcolor{RedNo}{Debilidades}}
      \small
      \begin{itemize}
        \item \highlight{Sobre-predicción sistemática} (Precision $0{,}59$--$0{,}70$): muchas falsas alarmas.
        \item No modela crecimiento \emph{leapfrog} (saltos discontinuos).
        \item Clase ``urbano'' incluye suelo desnudo/árido (RGB sin NIR).
        \item Omite variables externas (vialidades, empleo, pendiente).
      \end{itemize}
    \end{column}
  \end{columns}
  \vspace{0.1cm}
  \centering\footnotesize El error se origina en la clasificación y se \highlight{amplifica} por el efecto de vecindad del AC.
\end{frame}

  \begin{block}{}
  \begin{tabularx}{\linewidth}{@{}cl>{\raggedright\arraybackslash}X@{}}
    \toprule

    \midrule
    H1 & \textcolor{GreenOK}{\textbf{Confirmada}} & FoM $>0{,}10$ en \textbf{5/5} ventanas (rango $0{,}222$--$0{,}378$, prom.\ $0{,}317$ $\approx 3\times$ el piso); Kappa $0{,}445$ $\Rightarrow$ la evidencia espacial histórica es predictiva. \\
    H2 & \textcolor{GreenOK}{\textbf{Confirmada}} & Config.\ calibrada ($\theta{=}0{,}75$, $\alpha{=}0{,}50$, IV) da FoM $0{,}222$--$0{,}378$; una config.\ previa ($\theta{\approx}0{,}60$) sobre-predecía con FoM $\approx 0{,}17$ $\Rightarrow$ calibrar es necesario. \\
    H3 & \textcolor{GreenOK}{\textbf{Confirmada}} & Distancia (IV $0{,}28$) + densidades concentran $\sim$75\% del IV; relación WoE--densidad monótona creciente $\Rightarrow$ difusión por contagio. \\
    H4 & \textcolor{Amber}{\textbf{Parcial}} & Estabilidad morfológica en las 5 ventanas; FoM píxel-exacto sensible al crecimiento neto del período (el \emph{leapfrog} no modelado degrada la métrica). \\
    H5 & \textcolor{Teal}{\textbf{Sustentada}} & FoM $31{,}7\%$ en banda intermedia (Pontius), comparable pese a usar \emph{solo} 7 variables del mapa, sin perder interpretabilidad (no tabulada en el Cap.~7). \\
    H6 & \textcolor{Teal}{\textbf{Sustentada}} & Pipeline íntegro en Python abierto + datos libres, $\sim$3\,h en CPU sin GPU (discutida en ``Accesibilidad computacional''). \\
    \bottomrule
  \end{tabularx}
  \end{block}
  \centering\footnotesize H1--H4 confirmadas formalmente; H5 y H6 sustentadas por la evidencia operativa del modelo.
\end{frame}

  \footnotesize
  \begin{enumerate}
    \item \highlight{Integración WoE--AC} para una ciudad intermedia mexicana con datos de acceso libre (válida y potencialmente \emph{publicable}).
